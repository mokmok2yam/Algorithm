import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int X = sc.nextInt();
        int line = 1; 
        int sum = 0;
        while (true) {
            if (X <= sum + line) {
                break;
            }
            sum += line;
            line++;
        }

        int pos = X - sum;
        if (line % 2 == 0) {
            System.out.println(pos + "/" + (line - pos + 1));
        } else {
            System.out.println((line - pos + 1) + "/" + pos);
        }
    }
