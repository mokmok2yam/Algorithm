import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.Collections;
import java.util.LinkedList;
import java.util.Queue;
import java.util.StringTokenizer;

public class Main {
	static boolean visit[];
	static ArrayList<Integer>[] li;
	static StringBuilder sb;

	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringTokenizer st = new StringTokenizer(br.readLine());
		sb = new StringBuilder();
		int N = Integer.parseInt(st.nextToken());// 정점의 갯수
		visit = new boolean[N + 1]; // 방문배열 생성
		int V = Integer.parseInt(st.nextToken()); // 간선의 갯수
		int start = Integer.parseInt(st.nextToken());// 시작할 정점
		li = new ArrayList[N + 1]; // 리스트를 담을 배열 생성
		// 리스트 초기화
		for (int i = 1; i <= N; i++) {
			li[i] = new ArrayList<Integer>();
		}
		// 간선 넣어주기 무방향이므로 둘다 넣어줘야함
		for (int i = 0; i < V; i++) {
			st = new StringTokenizer(br.readLine());
			int a = Integer.parseInt(st.nextToken());
			int b = Integer.parseInt(st.nextToken());
			li[a].add(b);
			li[b].add(a);
		}
		// 더 작은 정점 먼저 간다했으므로 sort실행
		for (int i = 1; i <= N; i++) {
			Collections.sort(li[i]);
		}
		// dfs실행
		dfs(start);
		sb.append("\n");
		// 방문 배열 초기화
		visit = new boolean[N + 1];
		// bfs실행
		bfs(start);
		System.out.println(sb);
	}

	private static void bfs(int n) {
		// 방문 tre로 설정
		visit[n] = true;
		// 큐 생성
		Queue<Integer> q = new LinkedList<>();
		q.add(n);
		// 큐가 빌때까지 실행
		while (!q.isEmpty()) {
			// q에서 값하나 꺼내서 sb에 넣어줌
			int node = q.poll();
			sb.append(node).append(" ");
			for (int next : li[node]) {
				// 방문하지 않았으면 실행
				if (!visit[next]) {
					visit[next] = true;
					q.add(next);
				}
			}
		}

	}

	private static void dfs(int n) {
		// 방문 배열 true로 설정
		visit[n] = true;
		sb.append(n).append(" ");

		for (int next : li[n]) {
			// 방문 하지않았으면 dfs실행
			if (!visit[next]) {
				dfs(next);
			}
		}

	}
}
