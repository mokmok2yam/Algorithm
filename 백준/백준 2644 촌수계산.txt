import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.ArrayList;
import java.util.LinkedList;
import java.util.Queue;
import java.util.StringTokenizer;

public class Main {
	static int distance[]; //거리배열
	static ArrayList<Integer>[] li; 
	static int end; // 찾아야하는 사람

	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		int N = Integer.parseInt(br.readLine());
		StringTokenizer st = new StringTokenizer(br.readLine());
		int start = Integer.parseInt(st.nextToken());//시작하는 사람
		end = Integer.parseInt(st.nextToken()); // 찾아야하는 사람
		int v = Integer.parseInt(br.readLine()); // 간선의 수
		distance = new int[N + 1];
		for (int i = 1; i <= N; i++) {
			distance[i] = -1; //거리 -1로 초기화
		}
		li = new ArrayList[N + 1];
		for (int i = 1; i <= N; i++) {
			li[i] = new ArrayList<Integer>(); 
		}
		for (int i = 0; i < v; i++) {
			st = new StringTokenizer(br.readLine()); 
			int a = Integer.parseInt(st.nextToken()); 
			int b = Integer.parseInt(st.nextToken());
			li[a].add(b); //간선추가
			li[b].add(a);//간선 추가
		}
		bfs(start); //bfs시작
		System.out.println(distance[end]);
	}

	public static void bfs(int n) {
		Queue<Integer> q = new LinkedList<>(); //큐 생성
		distance[n]=0; //시작하는점 distance 0으로 시작
		q.add(n);//큐에 집어넣기
		while (!q.isEmpty()) {
			int now = q.poll(); 
			for (int next : li[now]) {
				if (distance[next] == -1)//distance가 -1일때만 시행 {
					distance[next] = distance[now] + 1; //현재 거리에서 +1
					q.add(next);
				}
			}
		}

	}
}
