import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;
import java.util.StringTokenizer;

public class Main {
	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringBuilder sb = new StringBuilder();
		int N = Integer.parseInt(br.readLine());// 스위치수 입력받기
		boolean arr[] = new boolean[N + 1];
		StringTokenizer st = new StringTokenizer(br.readLine());
		for (int i = 1; i <= N; i++) {
			if (Integer.parseInt(st.nextToken()) == 1) {
				arr[i] = true;
			} else
				arr[i] = false;
		}
		int K = Integer.parseInt(br.readLine());// 학생수입력받기
		for (int i = 0; i < K; i++) {
			st = new StringTokenizer(br.readLine());
			int gender = Integer.parseInt(st.nextToken());// 성입력
			int number = Integer.parseInt(st.nextToken()); // 자연수
			// 남자일때
			if (gender == 1) {
				for (int j = 1; j <= N; j++) {
					if (j % number == 0) { //
						arr[j] = !arr[j];
					}
				}
			} else// 여자일때
			{
				arr[number] = !arr[number];// 우선 중앙값은 무조건 변경
				int j = 1;
				// 스위치 범위 벗어나면 종료
				while ((number - j >= 1) && (number + j <= N)) {
					if (arr[number - j] == arr[number + j]) // 대칭인경우
					{
						arr[number - j] = !arr[number - j]; // 대칭값들 not연산
						arr[number + j] = !arr[number + j]; // 대칭값들 not연산
						j++;
					} else
						break;
				}
			}

		}
		for (int i = 1; i <= N; i++) {
			if (arr[i]) {
				sb.append("1 ");
			} else {
				sb.append("0 ");
			}
			if (i % 20 == 0) {
				sb.append("\n");
			}
		}
		System.out.println(sb);
	}
}
