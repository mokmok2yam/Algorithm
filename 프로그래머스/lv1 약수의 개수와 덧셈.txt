// lv1 약수의 개수와 덧셈
class Solution {
    public int solution(int left, int right) {
        int sum =0;
        for(int i=left;i<=right;i++){
            if((getDivisor(i))%2==0){
                sum+=i;
            }
            else{
                sum-=i;
            }
        }
        return sum;
    }
    public static int getDivisor(int n){
        int total =0;
        for(int i=1;i<=n;i++){
            if( n %i==0){
                total++;
            }
        }
        return total;
    }
}
