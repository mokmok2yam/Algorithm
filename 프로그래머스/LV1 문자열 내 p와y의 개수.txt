//문자열 내 p와y의 개수
class Solution {
    boolean solution(String s) {
        s = s.toUpperCase();
        int pCount=0;
        int yCount=0;
        char [] arr = s.toCharArray();
        for(int i=0;i<arr.length;i++){
            if(arr[i]-'P'==0){
                pCount++;
            }
            else if(arr[i]-'Y'==0){
                yCount++;
            }
        }
        return pCount==yCount;
    }
}
