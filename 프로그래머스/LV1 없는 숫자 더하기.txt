//LV1 없는 숫자 더하기
class Solution {
    public int solution(int[] numbers) {
        boolean visit[] = new boolean[10];
        for(int num:numbers){
            visit[num]=true;
        }
        int sum =0;
        for(int i=0;i<10;i++){
            if(!visit[i]){
                sum+=i;
            }
        }
        return sum;
    }
}
