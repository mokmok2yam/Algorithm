//lv1 문자열 다루기 기본
class Solution {
    public boolean solution(String s) {
        boolean isTrue = true;
        char [] ch = s.toCharArray();
        if(!(ch.length==4 || ch.length==6)){
            return false;
        }
        for(char c : ch){
            if(!(c-'0'>=0 && c-'0'<=9)){
                isTrue=false;
            }
        }
        return isTrue;
    }
}
