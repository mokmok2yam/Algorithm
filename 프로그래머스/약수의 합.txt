import java.util.*;
class Solution {
    public int solution(int n) {
        int sum =0 ;
        for(int i=1;i<=(int)Math.sqrt(n);i++){
            if(n%i==0){
                if(i==n/i){
                    sum+=i;
                }
                else{
                    sum+=i+n/i;
                }
                
            }
        }
        return sum;
        
    }
}
