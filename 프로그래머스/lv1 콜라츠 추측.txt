//lv1 콜라츠 추측
class Solution {
    public int solution(int num) {
        long n = num;
        int count =0;
        while(count<=500){
            if(n==1){
                break;
            }
            n = (n%2==0) ? n/2  : n*3+1;
            count++;
        }
        return n==1? count:-1;
    }
}
