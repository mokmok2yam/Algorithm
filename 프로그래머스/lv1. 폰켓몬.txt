lv1. 폰켓몬
import java.util.*;
class Solution {
    public int solution(int[] nums) {
        Set<Integer> set = new HashSet<>();
        for(int num : nums){
            set.add(num);
        }
        int max = nums.length / 2;
        int size = set.size();
        return size<max ? size : max;
    }
}
